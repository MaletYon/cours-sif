\chapter{Bibliographie Commentée}
\label{ann:bibliographie}

Pour chaque partie, quelques lectures, avec ce qu'elles apportent. Les références
complètes figurent dans la bibliographie du livre.

\section*{Pour commencer}
\begin{itemize}
    \item R.~Anderson, \emph{Security Engineering}~\cite{anderson2020}~: la référence pour comprendre comment les systèmes échouent.
    \item J.~Saltzer et M.~Schroeder~\cite{saltzer1975}~: les principes de conception, toujours actuels.
\end{itemize}

\section*{Partie I · Fondements}
\begin{itemize}
    \item J.~Katz et Y.~Lindell~\cite{katz2020}~: la cryptographie moderne, rigoureuse.
    \item W.~Diffie et M.~Hellman~\cite{diffie1976}, R.~Rivest, A.~Shamir et L.~Adleman~\cite{rivest1978}~: les articles fondateurs.
    \item E.~Spafford~\cite{spafford1989}, N.~Falliere et al.~\cite{falliere2011}~: deux analyses d'attaques qui ont fait date.
\end{itemize}

\section*{Partie II · Gouvernance et risques}
\begin{itemize}
    \item ISO/IEC~27014 et 27005~\cite{iso27014_2020,iso27005_2022}~; ANSSI, EBIOS RM~\cite{anssi2018ebios}.
    \item R.~Cialdini~\cite{cialdini1984}~; D.~Cappelli et al.~\cite{cappelli2012}~: l'humain et la menace interne.
\end{itemize}

\section*{Partie III · Normes et droit}
\begin{itemize}
    \item ISO/IEC~27001 et 27002~\cite{iso27001_2022,iso27002_2022}~; NIST CSF~2.0~\cite{nist2024csf}~; CIS Controls~\cite{cis2021controls}.
    \item Loi n°~2010/012~\cite{loi2010012}~; Convention de Malabo~\cite{malabo2014}.
\end{itemize}

\section*{Partie IV · Sécurité technique}
\begin{itemize}
    \item NIST SP~800-207 (zéro confiance)~\cite{nist800207}~; SP~800-63B (authentification)~\cite{nist80063b}.
    \item A.~Shostack~\cite{shostack2014}~; OWASP Top~10 et ASVS~\cite{owasp2021top10,owasp2021asvs}.
    \item NIST SP~800-61~\cite{nist80061r2,nist80061r3}~: la réponse aux incidents.
\end{itemize}

\section*{Parties V et VI · Défis et mesure}
\begin{itemize}
    \item ISO~22301~\cite{iso22301_2019}~; RFC~3227~\cite{rfc3227}~; NIST AI~100-2~\cite{nist2023aml}.
    \item ISO/IEC~27004~\cite{iso27004_2016}~; l'instrument \OUI{}~/ \NON{}~/ \NEK{}~\cite{minka2026nek}.
\end{itemize}
